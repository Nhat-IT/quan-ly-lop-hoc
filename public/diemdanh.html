// [NÂNG CẤP] Hàm đọc file thông minh (Smart Detection) function handleFileSelect(e) { const f = e.target.files[0]; if (!f) return; const r = new FileReader(); r.onload = function(ev) { try { const wb = XLSX.read(ev.target.result, {type: 'array'}); const ws = wb.Sheets[wb.SheetNames[0]]; // Lấy dữ liệu thô, giữ nguyên tiêu đề cột const jsonData = XLSX.utils.sheet_to_json(ws, {defval: ""}); if (jsonData.length === 0) { alert("File rỗng hoặc không đọc được dữ liệu!"); return; } // Lấy danh sách tiêu đề các cột có trong file const headers = Object.keys(jsonData[0]); // --- BỘ TỪ KHÓA NHẬN DIỆN (Viết thường để so sánh) --- const keys = { mssv: ['mssv', 'mã sv', 'mã sinh viên', 'mã số sinh viên', 'student id', 'code', 'masv'], fullName: ['họ và tên', 'họ tên', 'họ và tên sinh viên', 'tên sinh viên', 'full name', 'fullname', 'tên'], firstName: ['tên', 'first name', 'name'], // Trường hợp tách cột lastName: ['họ', 'họ lót', 'họ đệm', 'last name', 'surname'], // Trường hợp tách cột className: ['lớp', 'class', 'mã lớp', 'tên lớp', 'lớp học'] }; // Hàm tìm tên cột khớp với từ khóa const findHeader = (keywords) => { return headers.find(h => { const normalizedHeader = h.toLowerCase().trim(); return keywords.some(k => normalizedHeader.includes(k)); }); }; // 1. Dò cột MSSV const colMSSV = findHeader(keys.mssv); // 2. Dò cột Tên (Ưu tiên cột "Họ và tên" gộp, nếu không có thì tìm Họ + Tên riêng) let colFullName = findHeader(keys.fullName); let colLastName = null; let colFirstName = null; // Nếu không thấy cột "Họ tên" gộp, thử tìm cột "Họ" và "Tên" riêng if (!colFullName) { colLastName = findHeader(keys.lastName); colFirstName = findHeader(keys.firstName); } // 3. Dò cột Lớp const colClass = findHeader(keys.className); // --- KIỂM TRA KẾT QUẢ DÒ TÌM --- if (!colMSSV) { alert(`Lỗi: Không tìm thấy cột Mã sinh viên.\n(Code đã dò các từ khóa: ${keys.mssv.join(', ')})`); return; } if (!colFullName && (!colLastName || !colFirstName)) { alert(`Lỗi: Không tìm thấy cột Họ tên.\n(Code đã dò: ${keys.fullName.join(', ')})`); return; } // --- MAPPING DỮ LIỆU --- const sts = jsonData.map(row => { let nameFinal = ""; if (colFullName) { // Trường hợp 1: Có cột Họ tên gộp nameFinal = row[colFullName]; } else if (colLastName && colFirstName) { // Trường hợp 2: Ghép cột Họ + cột Tên nameFinal = `${row[colLastName]} ${row[colFirstName]}`; } return { mssv: row[colMSSV] ? String(row[colMSSV]).trim() : '', full_name: nameFinal ? String(nameFinal).trim() : '', class_name: colClass && row[colClass] ? String(row[colClass]).trim() : '' }; }).filter(s => s.mssv && s.full_name); // Lọc bỏ dòng lỗi // --- XÁC NHẬN VÀ GỬI --- if (sts.length === 0) { alert("Không trích xuất được sinh viên nào. Vui lòng kiểm tra nội dung file."); } else { let confirmMsg = `Đã tìm thấy ${sts.length} sinh viên.\n\n`; confirmMsg += `✔️ Cột MSSV: "${colMSSV}"\n`; confirmMsg += colFullName ? `✔️ Cột Tên: "${colFullName}"\n` : `✔️ Cột Tên: Ghép từ "${colLastName}" + "${colFirstName}"\n`; confirmMsg += colClass ? `✔️ Cột Lớp: "${colClass}"` : `⚠️ Cột Lớp: Không thấy (sẽ để trống)`; if(confirm(confirmMsg)) { sendImport(sts); } } } catch (err) { console.error(err); alert("Lỗi đọc file: " + err.message); } }; r.readAsArrayBuffer(f); e.target.value = ''; // Reset input để chọn lại file khác nếu cần }
